const PrintButton = ({ receiptRef }) => {
  const handlePrint = useReactToPrint({
    content: () => receiptRef.current,
    documentTitle: "Pharmacy Receipt",
  });

  return (
    <button
      style={{
        width: "100%",
        padding: "10px",
        marginTop: "10px",
        backgroundColor: "#007bff",
        color: "white",
        border: "none",
        cursor: "pointer",
      }}
      onClick={handlePrint}
    >
      Print Receipt
    </button>
  );
};
